so , 32 bit computer is 2^32 addresses, 0 pointer will point at memory location 0 bytes/0bits , 
1 poitner will point to memory location 1 byte/8bit

```cpp
#include <stdio.h>
#include <limits.h> // Обов'язково додаємо для CHAR_BIT

int main() {
    // Створюємо масив символів (кожен елемент займає 1 байт)
    char data[2] = {'A', 'B'};
    
    // Створюємо покажчик на перший елемент
    char *ptr = &data[0];
    
    printf("--- Перевірка адресації (крок в 1 байт) ---\n");
    printf("Початкова адреса (ptr):   %p\n", (void*)ptr);
    
    // Інкрементуємо покажчик на +1 char/+1 byte/+8 bits (move with 4 bits e.g impossible) 
    ptr = ptr + 1;
    
    printf("Адреса після ptr + 1:    %p\n", (void*)ptr);
    printf("Розмір покажчика у С:     %zu байтів\n", sizeof(ptr));
    printf("Розмір char у С:     %zu байтів\n", sizeof(char));

    // --- АВТОМАТИЧНЕ ВИЗНАЧЕННЯ АРХІТЕКТУРИ ---
    printf("\n--- Результат визначення системи ---\n");
    if (sizeof(ptr) == 8) {
        printf("Програма працює на: 64-бітній машині (архітектура x86_64 / ARM64)\n");
    } else if (sizeof(ptr) == 4) {
        printf("Програма працює на: 32-бітній машині (архітектура x86 / ARM32)\n");
    } else {
        printf("Невідома архітектура (Розмір покажчика: %zu байтів)\n", sizeof(ptr));
    }
    printf("Точна кількість бітів в одному байті (char): %d бітів\n", CHAR_BIT);

    return 0;
}
```
