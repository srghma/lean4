## Encoding `test_work` into lean4lean

The `match` desugars to `@Eq.rec α a (fun b _ => P b) hP b h`.

### Approach 1 — Elaborate the surface syntax and feed to lean4lean

The easiest route: let Lean's elaborator produce the `Expr` values, then hand
them to the lean4lean kernel as a `Declaration.thmDecl`.
-/

section TestWork_Elaboration

open Lean Meta

-- Elaborate the type and proof term at meta-level, then submit to lean4lean.
run_meta do
  -- Build the type: ∀ (α : Type) (P : α → Prop) (a b : α), a = b → P a → P b
  let ty ← Elab.Term.TermElabM.run' do
    Elab.Term.elabTerm (← `(term|
      ∀ (α : Type) (P : α → Prop) (a b : α) (h : Eq a b) (hP : P a), P b)) none
  -- Build the proof term: @Eq.rec α a (fun b _ => P b) hP b h
  let val ← Elab.Term.TermElabM.run' do
    Elab.Term.elabTerm (← `(term|
      fun (α : Type) (P : α → Prop) (a b : α) (h : Eq a b) (hP : P a) =>
        @Eq.rec α a (fun b _ => P b) hP b h)) (some ty)

/-!
### Approach 2 — Manual `Expr` construction

Build the full `Expr` tree by hand using de Bruijn indices.

\```
type:
  ∀ (α : Type)         -- Sort 1
    (P : α → Prop)     -- bvar 0 → Sort 0
    (a : α)            -- bvar 1
    (b : α)            -- bvar 2
    (h : Eq α a b)     -- Eq (bvar 3) (bvar 2) (bvar 0)
    (hP : P a),        -- (bvar 4) (bvar 3)
    P b                -- (bvar 5) (bvar 3)

value (inside 6 lambdas, bvars: 0=hP, 1=h, 2=b, 3=a, 4=P, 5=α):
  @Eq.rec.{1,0}
      (α    := bvar 5)
      (a    := bvar 3)
      (motive := fun b' (_ : Eq α a b') => P b')
      (h    := bvar 0)       -- hP  : P a  (the minor premise)
      (b    := bvar 2)
      (t    := bvar 1)       -- h   : a = b
\```
-/

section TestWork_Manual

open Lean

-- Binder-by-binder de Bruijn accounting (domain is at outer depth, body at inner depth):
--
--  depth 0 → 1 : forallE α (sort 1)             [body: bvar 0=α]
--  depth 1 → 2 : forallE P (bvar 0 → Prop)      [body: bvar 0=P, 1=α]
--  depth 2 → 3 : forallE a (bvar 1)              [body: bvar 0=a, 1=P, 2=α]
--  depth 3 → 4 : forallE b (bvar 2)              [body: bvar 0=b, 1=a, 2=P, 3=α]
--  depth 4 → 5 : forallE h  (Eq (bvar 3)(bvar 1)(bvar 0))  [body: bvar 0=h,1=b,2=a,3=P,4=α]
--  depth 5 → 6 : forallE hP (P a = (bvar 3)(bvar 2))       [body: bvar 0=hP,1=h,2=b,3=a,4=P,5=α]
--  depth 6     : return type = P b = (bvar 4)(bvar 2)
private def testWorkType : Expr :=
  .forallE `α (.sort 1)
  (.forallE `P (.forallE `_ (.bvar 0) (.sort 0) .default)
  (.forallE `a (.bvar 1)
  (.forallE `b (.bvar 2)
  (.forallE `h (mkApp3 (.const ``Eq [.succ .zero]) (.bvar 3) (.bvar 1) (.bvar 0))
  (.forallE `hP (.app (.bvar 3) (.bvar 2))   -- P a: depth-5 body, P=bvar 3, a=bvar 2
  (.app (.bvar 4) (.bvar 2))                  -- P b: depth-6 body, P=bvar 4, b=bvar 2
  .default) .default) .default) .default) .default) .default


-- The motive is embedded inside 6 outer lambdas (depth 6 context):
--   bvar 5=α, bvar 4=P, bvar 3=a, bvar 2=b, bvar 1=h, bvar 0=hP
--
-- lam b' : domain at depth 6  → α = bvar 5
--   depth 7: bvar 0=b', bvar 1=hP, bvar 2=h, bvar 3=b, bvar 4=a, bvar 5=P, bvar 6=α
-- lam _  : domain at depth 7  → Eq α a b' = Eq (bvar 6)(bvar 4)(bvar 0)
--   depth 8: bvar 0=_, bvar 1=b', bvar 2=hP, bvar 3=h, bvar 4=b, bvar 5=a, bvar 6=P, bvar 7=α
--   body = P b' = (bvar 6)(bvar 1)
private def testWorkMotiveFixed : Expr :=
  .lam `b' (.bvar 5)              -- b' : α  (α = bvar 5 at depth 6)
  (.lam `_
    (mkApp3 (.const ``Eq [.succ .zero]) (.bvar 6) (.bvar 4) (.bvar 0))
    -- Eq α a b'  at depth 7: α=bvar 6, a=bvar 4, b'=bvar 0
    (.app (.bvar 6) (.bvar 1))
    -- P b'       at depth 8: P=bvar 6, b'=bvar 1
    .default)
  .default


-- Binder-by-binder de Bruijn for the value (same as type, but lam instead of forallE):
--
--  depth 0 → 1 : lam α (sort 1)               [body: bvar 0=α]
--  depth 1 → 2 : lam P (bvar 0 → Prop)        [body: bvar 0=P, 1=α]
--  depth 2 → 3 : lam a (bvar 1)               [body: bvar 0=a, 1=P, 2=α]
--  depth 3 → 4 : lam b (bvar 2)               [body: bvar 0=b, 1=a, 2=P, 3=α]
--  depth 4 → 5 : lam h (Eq (bvar 3)(bvar 1)(bvar 0))    [body: bvar 0=h,1=b,2=a,3=P,4=α]
--  depth 5 → 6 : lam hP ((bvar 3)(bvar 2) = P a)        [body: bvar 0=hP,1=h,2=b,3=a,4=P,5=α]
--  depth 6     : Eq.rec call, α=bvar 5, a=bvar 3, hP=bvar 0, b=bvar 2, h=bvar 1
private def testWorkValue : Expr :=
  .lam `α (.sort 1)
  (.lam `P (.forallE `_ (.bvar 0) (.sort 0) .default)
  (.lam `a (.bvar 1)
  (.lam `b (.bvar 2)
  (.lam `h (mkApp3 (.const ``Eq [.succ .zero]) (.bvar 3) (.bvar 1) (.bvar 0))
  (.lam `hP (.app (.bvar 3) (.bvar 2))         -- P a: depth-5 domain, P=bvar 3, a=bvar 2
    -- @Eq.rec.{0,1} α a motive hP b h
    -- u_1=0 (motive → Prop), u_2=1 (α : Type)
    (mkApp6 (.const ``Eq.rec [.zero, .succ .zero])
      (.bvar 5)               -- α (depth-6 body)
      (.bvar 3)               -- a
      testWorkMotiveFixed     -- fun b' _ => P b' (open bvars relative to depth 6)
      (.bvar 0)               -- hP (minor premise: motive a rfl)
      (.bvar 2)               -- b
      (.bvar 1))              -- h  (major premise: a = b)
  .default) .default) .default) .default) .default) .default

run_meta do
  let kenv := (← getEnv).toKernelEnv
  let decl : Declaration := .thmDecl {
    name        := `test_work_manual
    levelParams := []
    type        := testWorkType
    value       := testWorkValue
  }
  match Lean4Lean.addDecl kenv decl with
  | .ok _  => logInfo "test_work_manual accepted by lean4lean ✓"
  | .error e => throwError "lean4lean rejected test_work_manual: {← (e.toMessageData {}).toString}"

end TestWork_Manual
